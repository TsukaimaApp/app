import React from 'react';
import { Icon } from '../core/Icon.jsx';
const AMP = [0.3, 0.65, 1, 0.65, 0.3];
export function VoiceIndicator({ state = 'listening', size = 160, style }) {
  const ring = Math.round(size * 0.62);
  const active = state === 'listening' || state === 'speaking';
  return (
    <div style={{ width: size, height: size, position: 'relative', display: 'flex', alignItems: 'center', justifyContent: 'center', flexShrink: 0, ...style }}>
      <style>{'@keyframes tsk-bar{0%,100%{transform:scaleY(.35)}50%{transform:scaleY(1)}}@keyframes tsk-ring{0%{transform:scale(.72);opacity:.7}100%{transform:scale(1.3);opacity:0}}@keyframes tsk-breathe{0%,100%{transform:scale(.96)}50%{transform:scale(1.04)}}@keyframes tsk-pulse{0%,100%{opacity:1}50%{opacity:.35}}'}</style>
      {state === 'speaking' && [0, 1].map(i => <span key={i} style={{ position: 'absolute', width: ring, height: ring, borderRadius: '50%', border: '1.5px solid var(--accent-selected)', animation: `tsk-ring 2.4s ${i * 1.2}s var(--ease-quiet) infinite` }}></span>)}
      <div style={{ width: ring, height: ring, borderRadius: '50%', background: state === 'muted' ? 'var(--surface-sunken)' : 'var(--accent-wash)', display: 'flex', alignItems: 'center', justifyContent: 'center', gap: Math.max(3, Math.round(size * 0.035)), animation: state === 'thinking' ? 'tsk-breathe 2.4s var(--ease-quiet) infinite' : 'none' }}>
        {state === 'muted' && <Icon name="mic_off" size={Math.round(size * 0.2)} color="var(--text-tertiary)" />}
        {state === 'idle' && <Icon name="mic" size={Math.round(size * 0.2)} color="var(--accent)" />}
        {state === 'thinking' && <span style={{ width: Math.round(size * 0.07), height: Math.round(size * 0.07), borderRadius: 99, background: 'var(--accent)', animation: 'tsk-pulse 2s infinite' }}></span>}
        {active && AMP.map((a, i) => <span key={i} style={{ width: Math.max(3, Math.round(size * 0.04)), height: Math.round(ring * (0.18 + a * 0.34)), borderRadius: 99, background: 'var(--accent)', transformOrigin: 'center', animation: `tsk-bar ${state === 'speaking' ? 0.9 : 1.4}s ${i * 0.13}s ease-in-out infinite` }}></span>)}
      </div>
    </div>
  );
}
